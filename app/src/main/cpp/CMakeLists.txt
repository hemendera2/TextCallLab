cmake_minimum_required(VERSION 3.22.1)
project(callcompanion_llm LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(GGML_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)

set(LLAMA_CPP_PATH "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_CPP_PATH}/CMakeLists.txt")
    message(FATAL_ERROR "Pinned llama.cpp source missing. CI must fetch before Gradle runs.")
endif()

# Static CPU llama.cpp library is bundled into our own JNI shared library.
add_subdirectory("${LLAMA_CPP_PATH}" "${CMAKE_CURRENT_BINARY_DIR}/llama-build" EXCLUDE_FROM_ALL)
add_library(callcompanion_llm SHARED callcompanion-llm.cpp)
target_compile_options(callcompanion_llm PRIVATE -O3 -fexceptions -frtti)
target_include_directories(callcompanion_llm PRIVATE "${LLAMA_CPP_PATH}/include")
target_link_libraries(callcompanion_llm PRIVATE llama log android)
